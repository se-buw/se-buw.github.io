{{- range $pages }}{{ partial "publication/entry.html" . }}{{ end -}}
{{- else -}} {{- $cards := "" -}} {{- range $pages -}} {{- $image := "" -}} {{- with .Resources.GetMatch "featured*" }}{{ $image = (.Fill "800x450 Smart webp").RelPermalink }}{{ end -}} {{- $cards = printf "%s%s" $cards (partial "shortcodes/card.html" (dict "link" .RelPermalink "title" .Title "subtitle" (.Params.summary | default .Summary | plainify) "image" $image "tag" (partial "utils/format-date" .Date) )) -}} {{- end -}} {{- partial "shortcodes/cards.html" (dict "cols" (len $pages) "content" ($cards | safeHTML)) -}} {{- end -}}
