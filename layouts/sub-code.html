{{- $file := .Get "file" -}} {{- $lang := .Get "lang" | default "typescript" -}} {{- /* Read the file from the local file system (submodule) */ -}} {{- $content := os.ReadFile $file -}} {{- /* Normalize line endings to avoid issues with Windows (\r\n) */ -}} {{- $content = replace $content "\r\n" "\n" -}} {{- $lines := split $content "\n" -}} {{- $totalLines := len $lines -}} {{- /* Get start and end lines, defaulting to the whole file */ -}} {{- $start := .Get "start" | default 1 -}} {{- $end := .Get "end" | default $totalLines -}} {{- /* Calculate the slice of lines to extract */ -}} {{- $startIdx := sub (int $start) 1 -}} {{- $length := sub (int $end) $startIdx -}} {{- $snippet := delimit (first $length (after $startIdx $lines)) "\n" -}} {{- /* Build the Chroma syntax highlighting options */ -}} {{- $options := slice -}} {{- /* Configure Line Numbers (default to true, starting at the extracted line) */ -}} {{- $showLineNos := .Get "linenos" | default "true" -}} {{- if ne $showLineNos "false" -}} {{- $options = $options | append "linenos=table" -}} {{- $options = $options | append (printf "linenostart=%v" $start) -}} {{- end -}} {{- /* Highlight specific lines */ -}} {{- with .Get "hl_lines" -}} {{- $options = $options | append (printf "hl_lines=%s" .) -}} {{- end -}} {{- $optsString := delimit $options ", " -}} {{- /* Render the code block */ -}} {{ highlight $snippet $lang $optsString }}
